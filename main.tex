\section{Heading: Section 4}\label{sec:four}

\begin{definition}[Consciousness]\label{def:consciousness}
  A creature is \emph{conscious} just in case there is something it is like to
  be that creature: its mental life has a subjective character, a way it seems
  to the creature itself. A creature with no such character would be a mere
  information processor, whatever it could do \autocite[435--450]{nagel1974}.
\end{definition}

\begin{definition}[Realisation]\label{def:realise}
  Let $M = \langle S, \delta, s_0, I, O \rangle$ be a deterministic machine
  with state set $S$, transition function $\delta \colon S \times I \to S$,
  initial state $s_0$, inputs $I$ and outputs $O$, together with an output map
  $\omega \colon S \to O$. A physical system $P$ \emph{realises} $M$ if there
  is an injective mapping $f \colon S \to \mathrm{States}(P)$ such that for all
  $s \in S$ and $i \in I$, the evolution of $P$ from $f(s)$ on input $i$ ends
  in $f(\delta(s,i))$.
\end{definition}

\section{Heading: Section 6}\label{sec:six}
Possibility can be defined from necessity:
\begin{equation}\label{eq:duality}
	\Possibly\varphi \leftrightarrow \neg\Necessarily\neg\varphi
\end{equation}

\noindent \Cref{eq:duality} is a theorem of every normal modal logic.
